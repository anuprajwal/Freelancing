async function createLinkedAccount(doctor) {
  const payload = {
    type: "individual",

    email: doctor.user?.email,
    phone: doctor.user?.phone_number,

    legal_business_name:
      doctor.clinic_name ||
      doctor.user?.full_name ||

  };

  return await razorpay.accounts.create(payload);
}
